cmake_minimum_required(VERSION 3.6.3)

project(libcurl-download LANGUAGES C)

find_program(MAKE_EXE NAMES make)

if (DEFINED CMAKE_OSX_SYSROOT AND NOT CMAKE_OSX_SYSROOT STREQUAL "")
  set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -isysroot${CMAKE_OSX_SYSROOT}")
endif()

include(ExternalProject)

# NOTE: Curl dropped the CMAKE_USE_* option prefix in 7.77 in favour of CURL_USE_*.
# The names below are post-7.77, and are required by the pinned version;
# they are silently ignored by curl < 7.77, so the option names and GIT_TAG must be
# kept in step.
set(CMAKE_ARGS
        -DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}
        -DCMAKE_INSTALL_PREFIX=${OPEN_SRC_INSTALL_PREFIX}
        -DBUILD_CURL_EXE=OFF
        -DCURL_USE_LIBSSH2=OFF
        -DBUILD_TESTING=OFF
        # Curl's picky warning set includes -Werror=partial-availability and similar,
        # which lets a dependency's warning policy break this build. Applies to every
        # toolchain, so it is set unconditionally.
        -DPICKY_COMPILER=OFF
        # Optional features this SDK does not use. Curl 8.x probes the host for them by
        # default, which makes the build depend on what happens to be installed, so they
        # are disabled explicitly. HTTP_ONLY does not cover them: they are not protocols.
        -DCURL_USE_LIBPSL=OFF
        -DUSE_NGHTTP2=OFF
        -DUSE_LIBIDN2=OFF
        -DCURL_BROTLI=OFF
        -DCURL_ZSTD=OFF
        -DHTTP_ONLY=ON)

# Apple only: pipe2() is macOS 27+, but check_function_exists() uses its own prototype
# and misses the availability annotation, so HAVE_PIPE2 is set even when the deployment
# target is older and clang then rejects the call in socketpair.c. Any SDK newer than the
# host OS hits this. OFF takes the pipe()+fcntl(FD_CLOEXEC) fallback. Not global: on
# Linux pipe2() is real and preferable, since it sets O_CLOEXEC atomically.
if (APPLE)
    set(CMAKE_ARGS ${CMAKE_ARGS} -DHAVE_PIPE2=OFF)
endif()

# Note: Future Curl versions introduce BUILD_STATIC_LIBS flag
if (BUILD_STATIC)
    set(CMAKE_ARGS ${CMAKE_ARGS} -DBUILD_SHARED_LIBS=OFF)
else()
    set(CMAKE_ARGS ${CMAKE_ARGS} -DBUILD_SHARED_LIBS=ON)
endif()

if (CURL_USE_FPIC)
    set(CMAKE_ARGS ${CMAKE_ARGS} -DCMAKE_POSITION_INDEPENDENT_CODE=ON)
endif()

# By default use openssl. Both backends must be selected explicitly: CURL_USE_MBEDTLS
# and CURL_USE_OPENSSL are cmake_dependent_option and default OFF, so omitting them
# leaves curl to pick a backend on its own.
if (USE_MBEDTLS)
    set(CMAKE_ARGS ${CMAKE_ARGS} -DCURL_USE_MBEDTLS=ON)
else()
    set(CMAKE_ARGS ${CMAKE_ARGS} -DCURL_USE_OPENSSL=ON)
endif()

# Tell Curl to find all its deps (OpenSSL/MbedTLS) in open-source directory
# (except when using pre-installed)
if (NOT OPENSSL_ROOT_DIR)
    set(CMAKE_ARGS
        ${CMAKE_ARGS}
        -DCMAKE_PREFIX_PATH=${OPEN_SRC_INSTALL_PREFIX}
        -DCMAKE_INSTALL_RPATH=$ORIGIN
    )
else()
    set(CMAKE_ARGS
        ${CMAKE_ARGS}
        -DCMAKE_INSTALL_RPATH=${OPENSSL_ROOT_DIR}/lib:$ORIGIN
    )
endif()

if (WIN32)
    # Schannel supersedes the OpenSSL selection made above on Windows.
    set(CMAKE_ARGS ${CMAKE_ARGS} -DCURL_USE_SCHANNEL=ON -DCURL_USE_OPENSSL=OFF -DCURL_STATIC_CRT=1)
endif()

# NOTE: curl 8.18.0 and later require OpenSSL 3.0 (lib/vtls/openssl.c). This project
# pins OpenSSL 1.1.1 in libopenssl-CMakeLists.txt, so 8.17.0 is the highest usable
# version until that pin moves. Do not bump past 8.17.0 in isolation.
ExternalProject_Add(project_libcurl
    GIT_REPOSITORY    https://github.com/curl/curl.git
    GIT_TAG           curl-8_17_0
    GIT_SHALLOW       TRUE
    GIT_PROGRESS      TRUE
    PREFIX            ${CMAKE_CURRENT_BINARY_DIR}/build
    CMAKE_ARGS        ${CMAKE_ARGS}
    BUILD_ALWAYS      TRUE
    TEST_COMMAND      "")
