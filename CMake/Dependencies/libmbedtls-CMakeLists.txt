cmake_minimum_required(VERSION 3.6.3)

project(libmbedtls-download LANGUAGES C)

include(ExternalProject)

set(BUILD_SHARED 1)
if (BUILD_STATIC)
	set(BUILD_SHARED 0)
endif()

set(CMAKE_MACOSX_RPATH 0)
if (${CMAKE_SYSTEM_NAME} MATCHES "Darwin")
    set(CMAKE_MACOSX_RPATH 1)
endif()

# curl 8.17.0 (libcurl-CMakeLists.txt) requires mbedTLS >= 3.2.0, enforced both at curl's
# configure step and with a #error in lib/vtls/mbedtls.c.
# USE_LEGACY_MBEDTLS (deprecated) selects the last 2.x LTS instead; libcurl-CMakeLists.txt
# pairs it with curl 8.15.0, the last curl release that accepts mbedTLS 2.x.
if (USE_LEGACY_MBEDTLS)
  set(MBEDTLS_GIT_TAG mbedtls-2.28.10)
else()
  set(MBEDTLS_GIT_TAG mbedtls-3.6.7)
endif()

ExternalProject_Add(
  project_libmbedtls
  GIT_REPOSITORY 	https://github.com/Mbed-TLS/mbedtls.git
  GIT_TAG 			  ${MBEDTLS_GIT_TAG}
  PREFIX 			    ${CMAKE_CURRENT_BINARY_DIR}/build
  CMAKE_ARGS
  	-DCMAKE_INSTALL_PREFIX=${OPEN_SRC_INSTALL_PREFIX}
    -DUSE_SHARED_MBEDTLS_LIBRARY=${BUILD_SHARED}
    -DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}
    -DCMAKE_MACOSX_RPATH=${CMAKE_MACOSX_RPATH}
    # The SDK only links the libraries; building mbedTLS's sample programs and test suite
    # wastes build time and pulls a Python toolchain for test-data generation in 3.x.
    -DENABLE_PROGRAMS=OFF
    -DENABLE_TESTING=OFF
  BUILD_ALWAYS 		TRUE
  TEST_COMMAND 		"")
